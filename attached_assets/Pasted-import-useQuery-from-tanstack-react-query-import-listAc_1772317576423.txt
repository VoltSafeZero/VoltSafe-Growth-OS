import { useQuery } from "@tanstack/react-query";
import { listAccounts } from "./api";
import { Input } from "@/components/ui/input";
import { useState } from "react";
import { Link } from "wouter";

export function AccountsTable() {
  const [q, setQ] = useState("");
  const { data, isLoading, error } = useQuery({
    queryKey: ["accounts", q],
    queryFn: () => listAccounts(q),
  });

  if (isLoading) return <div className="text-sm text-muted-foreground">Loading…</div>;
  if (error) return <div className="text-sm text-red-400">{String(error)}</div>;

  return (
    <div className="space-y-3">
      <div className="flex gap-2">
        <Input
          value={q}
          onChange={(e) => setQ(e.target.value)}
          placeholder="Search marinas…"
          className="max-w-sm"
        />
      </div>

      <div className="rounded-2xl border border-white/10 overflow-hidden">
        <div className="grid grid-cols-12 px-4 py-2 text-xs text-muted-foreground bg-white/5">
          <div className="col-span-4">Marina</div>
          <div className="col-span-2">Status</div>
          <div className="col-span-2">Priority</div>
          <div className="col-span-2">Slips</div>
          <div className="col-span-2">Next Action</div>
        </div>

        {data?.map((a) => (
          <Link key={a.id} href={`/crm/accounts/${a.id}`}>
            <a className="grid grid-cols-12 px-4 py-3 text-sm hover:bg-white/5 border-t border-white/10">
              <div className="col-span-4">
                <div className="font-medium">{a.accountName}</div>
                <div className="text-xs text-muted-foreground">
                  {[a.city, a.stateProvince, a.country].filter(Boolean).join(", ")}
                </div>
              </div>
              <div className="col-span-2">{a.leadStatus}</div>
              <div className="col-span-2">{a.priority}</div>
              <div className="col-span-2">{a.slipCount ?? "–"}</div>
              <div className="col-span-2 truncate">{a.nextAction ?? "–"}</div>
            </a>
          </Link>
        ))}
      </div>
    </div>
  );
}
